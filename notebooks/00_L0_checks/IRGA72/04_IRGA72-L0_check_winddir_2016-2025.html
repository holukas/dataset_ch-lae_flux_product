col = 'WD'
wd = df[col].copy()

# The last run reaches a few half-hours into the next January, which would enter the analysis as a
# year of its own with almost no records in it.
LASTYEAR = 2025

# The reference year the yearly offsets are measured against is deliberately left at 2024, the year
# it was when this was last run: moving it would change every year's offset and make the results
# incomparable with the previous run, rather than adding a year to them.
REFYEARS = [2024]

# Prepare input data
wd = wd.loc[wd.index.year <= LASTYEAR]
wd = wd.dropna()

wds = WindDirOffset(winddir=wd, offset_start=-50, offset_end=50, hist_ref_years=REFYEARS, hist_n_bins=360)
yearlyoffsets_df = wds.get_yearly_offsets()
print(yearlyoffsets_df)
print(wd)
